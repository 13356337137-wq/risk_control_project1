### 只读覆盖检查完整命令（PowerShell）

```powershell
$contract = Get-Content -Raw -Encoding UTF8 contract.md
$dict = Get-Content -Raw -Encoding UTF8 data/data_dictionary.md
$section = ($contract -split '# 数据契约（十一张表）')[1] -split '# 状态机' | Select-Object -First 1
$tables = [regex]::Matches($section, '(?m)^\| ([a-z_][a-z0-9_]*) \| ([^|]+) \| ([^|]+) \|')
$blocks = [regex]::Matches($dict, '(?ms)^## 表：([a-z_][a-z0-9_]*)\r?\n(.*?)(?=^## |\z)')
$failed = $tables.Count -ne 11 -or $blocks.Count -ne 11
foreach ($table in $tables) {
  $name = $table.Groups[1].Value
  $expected = $table.Groups[2].Value.Trim() -split '、'
  $block = @($blocks | Where-Object { $_.Groups[1].Value -eq $name })
  $actual = @([regex]::Matches(($block | ForEach-Object { $_.Groups[2].Value }) -join '', '(?m)^\| ([a-z_][a-z0-9_]*) \|') | ForEach-Object { $_.Groups[1].Value })
  $missing = @($expected | Where-Object { $_ -cnotin $actual })
  $extra = @($actual | Where-Object { $_ -cnotin $expected })
  $duplicates = @($actual | Group-Object | Where-Object Count -gt 1 | ForEach-Object Name)
  if ($block.Count -ne 1 -or $missing.Count -or $extra.Count -or $duplicates.Count) { $failed = $true }
  "$name contract=$($expected.Count) dictionary=$($actual.Count) missing=[$($missing -join ',')] extra=[$($extra -join ',')] duplicates=[$($duplicates -join ',')]"
}
$unknown = @($blocks | Where-Object { $_.Groups[1].Value -cnotin @($tables | ForEach-Object { $_.Groups[1].Value }) })
if ($unknown.Count) { $failed = $true }
$chat = Get-Content -Raw -Encoding UTF8 data/chat.md
$questions = @([regex]::Matches($chat, '(?m)^- \[ \] \*\*(Q\d+) ') | ForEach-Object { $_.Groups[1].Value })
foreach ($i in 1..9) { $q = 'Q{0:D2}' -f $i; if (@($questions | Where-Object { $_ -eq $q }).Count -ne 1 -or $dict -notmatch "\*\*$q ") { $failed = $true } }
$task = Get-Content -Raw -Encoding UTF8 data/tasks.md
if ($task -notmatch '\| \[ \] \| DA-06 \| 数据字典 \| 字典与建表脚本字段一致 \|') { $failed = $true }
"tables: contract=$($tables.Count) dictionary=$($blocks.Count) unknown=$($unknown.Count)"
"questions=$($questions -join ','); DA-06 original unchecked row preserved"
if ($failed) { throw 'Coverage FAIL' } else { 'Coverage PASS' }
```
